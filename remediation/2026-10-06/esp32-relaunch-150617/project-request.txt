~~~cpp
void writeLed() {
  const uint8_t level = (ledOn == static_cast<bool>(LED_ACTIVE_HIGH)) ? HIGH : LOW;
  digitalWrite(LED_PIN, level);
}

void sendStatus() {
  char json[192];
  const unsigned long long seconds =
      static_cast<unsigned long long>(esp_timer_get_time() / 1000000LL);
  snprintf(json, sizeof(json),
           "{\"uptimeSeconds\":%llu,\"freeHeap\":%lu,\"clients\":%u,\"led\":%s}",
           seconds, static_cast<unsigned long>(ESP.getFreeHeap()),
           static_cast<unsigned int>(WiFi.softAPgetStationNum()),
           ledOn ? "true" : "false");
  server.sendHeader("Cache-Control", "no-store");
  server.send(200, "application/json", json);
}

void handlePage() {
  if (server.method() != HTTP_GET) {
    server.sendHeader("Allow", "GET");
    server.send(405, "text/plain", "Use GET");
    return;
  }
  server.send_P(200, "text/html; charset=utf-8", DASHBOARD_HTML);
}

void handleStatus() {
  if (server.method() != HTTP_GET) {
    server.sendHeader("Allow", "GET");
    server.send(405, "text/plain", "Use GET");
    return;
  }
  sendStatus();
}

void handleLed() {
  if (server.method() != HTTP_POST) {
    server.sendHeader("Allow", "POST");
    server.send(405, "text/plain", "Use POST");
    return;
  }
  String contentType = server.header("Content-Type");
  const int separator = contentType.indexOf(';');
  if (separator >= 0) contentType = contentType.substring(0, separator);
  contentType.trim();
  contentType.toLowerCase();
  if (contentType != "application/x-www-form-urlencoded") {
    server.send(415, "text/plain", "Use application/x-www-form-urlencoded");
    return;
  }
  if (server.args() != 1 || server.argName(0) != "state") {
    server.send(400, "text/plain", "Provide exactly one state parameter");
    return;
  }
  const String state = server.arg(0);
  if (state != "on" && state != "off") {
    server.send(400, "text/plain", "State must be on or off");
    return;
  }
  ledOn = state == "on";
  writeLed();
  sendStatus();
}

void setup() {
  Serial.begin(SERIAL_BAUD);
  ledOn = false;
  writeLed();
  pinMode(LED_PIN, OUTPUT);
  writeLed();

  if (!WiFi.mode(WIFI_AP) ||
      !WiFi.softAPConfig(AP_IP, AP_IP, AP_SUBNET) ||
      !WiFi.softAP(AP_SSID, AP_PASSWORD)) {
    Serial.println("Access point initialization failed. Restart to retry.");
    return;
  }

  server.collectHeaders(REQUEST_HEADERS, 1);
  server.on("/", handlePage);
  server.on("/dashboard", handlePage);
  server.on("/api/status", handleStatus);
  server.on("/led", handleLed);
  server.onNotFound([]() { server.send(404, "text/plain", "Not found"); });
  server.begin();
  serverReady = true;
  Serial.print("Dashboard ready at http://");
  Serial.println(WiFi.softAPIP());
  Serial.print("Wi-Fi network: ");
  Serial.println(AP_SSID);
}

void loop() {
  if (serverReady) server.handleClient();
  yield();
}
~~~

~~~markdown
# ESP32 Wi-Fi Dashboard

A local Wi-Fi dashboard for a standard ESP32 DevKit, using PlatformIO and the Arduino framework. It controls a configurable LED output and shows uptime, free memory and the number of clients connected to the ESP32 access point.

## Hardware

Use an ESP32 DevKit compatible with PlatformIO's esp32dev target. Power it through its USB connector. ESP32 GPIO uses 3.3 V logic; do not feed 5 V into a GPIO.

GPIO 2 is the default LED pin. LED availability and polarity vary by board. Change LED_PIN and LED_ACTIVE_HIGH in include/config.h to match yours. The default is active high, and the commanded state starts OFF.

For an optional external LED with active-high configuration, connect GPIO 2 through a 220–330 ohm resistor to the LED anode, and the cathode to GND. Use LED_ACTIVE_HIGH true. Do not connect the LED directly without a resistor.

## Build and upload

The project uses only libraries included with the ESP32 Arduino core. No lib_deps entry is needed.

- Compile: pio run -e esp32dev
- Upload when the intended ESP32 is connected: pio run -e esp32dev -t upload --upload-port YOUR_PORT
- Open serial output: pio device monitor -b 115200 -p YOUR_PORT

ForgeX's build view can compile the saved project. Review and apply its draft files before uploading. Select the port belonging to the ESP32, since Bluetooth COM ports do not identify an ESP32.

## Connect to the dashboard

After uploading and booting:

1. Join the Wi-Fi network ESP32-Dashboard.
2. Use the demo password esp32demo123.
3. Open http://192.168.4.1 in your browser. The /dashboard alias serves the same page.

The dashboard works offline, updates every second and refreshes after LED commands. Change the demo AP credentials in include/config.h before using it outside a test setup.

## HTTP API

GET /api/status returns JSON with uptimeSeconds, freeHeap, clients and a boolean led value. Clients counts Wi-Fi stations, including a phone or computer connected to the AP.

POST /led expects application/x-www-form-urlencoded with exactly one parameter: state=on or state=off. A successful response contains the updated status JSON. Invalid/missing/extra/duplicate parameters return 400. Wrong methods return 405. Unsupported content types return 415. Unknown routes return 404.

## Validation

A successful build confirms the firmware compiles for esp32dev. It does not confirm Wi-Fi behavior, LED wiring or operation on a physical board. Those require uploading to the intended ESP32 and testing the dashboard there.

The LED status shows the commanded output, not electrical feedback from the pin or LED. Uptime uses the ESP32 timer instead of a 32-bit millisecond counter.
~~~
